async function generatePlan() { const btn = document.getElementById('runBtn'); const output = document.getElementById('output'); const loading = document.getElementById('loading'); const teamLines = document.getElementById('team').value.split('\n'); const team_members = teamLines.map(line => { const [name, skills] = line.split('|'); return { name: name?.trim(), skills: skills?.trim() }; }); const data = { project_name: document.getElementById('projName').value, industry: document.getElementById('industry').value, requirements: document.getElementById('requirements').value, team_members: team_members }; btn.disabled = true; loading.style.display = 'block'; output.style.display = 'none'; try { const response = await fetch('/api/generate-plan', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) }); const result = await response.json(); if (response.ok) { let html = `

Final Project Plan: ${result.project}

`; // --- MERGING LOGIC: Group tasks by assigned person --- const groupedTasks = {}; result.plan.forEach(task => { const person = task.assigned_to; if (!groupedTasks[person]) { groupedTasks[person] = []; } groupedTasks[person].push(task); }); // --- GENERATING THE HTML SECTIONS --- for (const person in groupedTasks) { html += `

👤 Tasks for ${person}

`; groupedTasks[person].forEach(task => { html += `
✅ ${task.task_name}
⏱️ Hours: ${task.estimated_hours}

${task.rationale}

`; }); } output.innerHTML = html; } else { output.innerText = "Error: " + (result.detail || "Check Vercel Logs"); } } catch (err) { output.innerText = "Error: " + err.message; } finally { btn.disabled = false; loading.style.display = 'none'; output.style.display = 'block'; } }
